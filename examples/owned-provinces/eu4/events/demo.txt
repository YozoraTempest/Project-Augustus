country_event = {
    id = owned.start
    is_triggered_only = yes
    trigger = {
        any_owned_province = { has_province_modifier = local }
    }
    immediate = {
        if = {
            limit = { any_owned_province = { province_id = 3 } }
            log = found
        }
        if = {
            limit = { NOT = { any_owned_province = { province_id = 1 } } }
            log = foreign
        }
        capital_scope = {
            owner = {
                if = {
                    limit = { any_owned_province = { province_id = 3 } }
                    log = owner
                }
            }
        }
    }
    option = {
        name = continue
        3 = { remove_province_modifier = local }
        if = {
            limit = { NOT = { any_owned_province = { has_province_modifier = local } } }
            log = removed
        }
        2 = {
            set_owner = BBB
            owner = {
                if = {
                    limit = { any_owned_province = { province_id = 2 } }
                    log = transferred
                }
            }
        }
        country_event = { id = owned.later days = 1 }
    }
    after = {
        if = {
            limit = { NOT = { any_owned_province = { has_province_modifier = local } } }
            log = after
        }
    }
}

country_event = {
    id = owned.later
    is_triggered_only = yes
    trigger = {
        NOT = { any_owned_province = { has_province_modifier = local } }
    }
    immediate = { log = later }
    option = { name = finish }
}
