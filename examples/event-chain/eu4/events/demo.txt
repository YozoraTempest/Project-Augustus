namespace = augustus_council

country_event = {
    id = augustus_council.2
    is_triggered_only = yes
    immediate = { log = council_ready }
    option = {
        name = commission_harbor
        log = work_started
        country_event = { id = augustus_council.3 days = 2 }
        capital_scope = { province_event = { id = augustus_council.4 days = 0 } }
        log = work_planned
    }
    after = { log = council_closed }
}
